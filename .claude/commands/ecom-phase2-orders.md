### Étape 2 — Nouveaux fichiers à créer

**Migration V14** : `src/main/resources/db/migration/V14__evolve_orders_domain.sql`

```sql
-- 1. Add e-commerce columns to orders (nullable for backward compatibility)
ALTER TABLE orders
    ADD COLUMN user_id       BIGINT REFERENCES app_users(id),
    ADD COLUMN order_number  VARCHAR(50) UNIQUE,
    ADD COLUMN status        VARCHAR(20) NOT NULL DEFAULT 'PENDING',
    ADD COLUMN total_amount  NUMERIC(12,2);

-- 2. Create order_items table
CREATE TABLE order_items (
    id           BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    order_id     BIGINT NOT NULL REFERENCES orders(id),
    product_id   BIGINT REFERENCES products(id) ON DELETE SET NULL,
    product_name VARCHAR(255) NOT NULL,
    unit_price   NUMERIC(12,2) NOT NULL,
    quantity     INT NOT NULL CHECK (quantity > 0),
    subtotal     NUMERIC(12,2) NOT NULL
);

CREATE INDEX idx_order_items_order_id ON order_items(order_id);

-- 3. Migrate historical orders: create one OrderItem per existing order
--    Use product snapshot from current products table
INSERT INTO order_items (order_id, product_id, product_name, unit_price, quantity, subtotal)
SELECT
    o.id,
    o.product_id,
    p.product_name,
    CAST(p.unit_price AS NUMERIC(12,2)),
    o.quantity,
    ROUND(CAST(o.quantity AS NUMERIC) * CAST(p.unit_price AS NUMERIC(12,2)), 2)
FROM orders o
JOIN products p ON p.id = o.product_id;

-- 4. Backfill total_amount from historical total column
UPDATE orders SET total_amount = ROUND(CAST(total AS NUMERIC(12,2)), 2);

-- 5. Backfill order_number for historical orders
UPDATE orders SET order_number = 'ORD-HIST-' || LPAD(id::TEXT, 6, '0');

-- NOTE: user_id stays NULL for historical orders (no guaranteed Customer<->AppUser link).
--       product_id and customer_id columns are kept for backward compatibility.
```
